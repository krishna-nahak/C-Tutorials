#include <stdio.h>
#include <ctype.h>

int stack[100];
int top = -1;

void push(int x) {
    top++;
    stack[top] = x;
}

int pop() {
    int x = stack[top];

    top--;

    return x;
}

int main() {
    char exp[100];
    int i, a, b, ans;

    printf("Enter postfix expression: ");
    scanf("%s", exp);

    for (i = 0; exp[i] != '\0'; i++) {
        if (isdigit(exp[i])) {
            push(exp[i] - '0');
        }
        else {
            b = pop();
            a = pop();

            if (exp[i] == '+') {
                ans = a + b;
            }
            else if (exp[i] == '-') {
                ans = a - b;
            }
            else if (exp[i] == '*') {
                ans = a * b;
            }
            else if (exp[i] == '/') {
                ans = a / b;
            }

            push(ans);
        }
    }

    printf("Result = %d", pop());

    return 0;
}

struct Node* add(struct Node *p1, struct Node *p2)
{

while (p1 != NULL && p2 != NULL)
{

{

p2 = p2->next;
}
